cmake_minimum_required(VERSION 3.21)
project(OpenPFCAcceptance LANGUAGES C CXX)
set(backend cpu CACHE STRING "cpu, cuda or hip")
set_property(CACHE backend PROPERTY STRINGS cpu cuda hip)
if(backend STREQUAL "cuda")
  enable_language(CUDA)
elseif(backend STREQUAL "hip")
  enable_language(HIP)
elseif(NOT backend STREQUAL "cpu")
  message(FATAL_ERROR "backend must be cpu, cuda or hip")
endif()
if(backend STREQUAL "hip")
  # Cray's CXX wrapper may supply MPI implicitly. HIP compilation/linking
  # uses Clang, so require FindMPI's explicit wrapper includes and libraries.
  # FindMPI still compiles and links its normal checks.
  set(MPI_ASSUME_NO_BUILTIN_MPI TRUE)
endif()
find_package(OpenPFC 0.5 CONFIG REQUIRED)
string(TOUPPER "${backend}" _backend)
# REQUIRED verifies every host/MPI evidence check. The SDK API deliberately
# leaves GPU runtime compatibility unverified; the compiled device case and
# its explicit runtime preflight supply that evidence, never a CMake override.
openpfc_check_compatibility(BACKEND CPU REQUIRED RESULT _result REPORT _report)
if(NOT backend STREQUAL "cpu")
  openpfc_check_compatibility(BACKEND "${_backend}" RESULT _gpu_result REPORT _gpu_report)
  if(_gpu_result STREQUAL "incompatible")
    message(FATAL_ERROR "${_gpu_report}")
  endif()
  message(STATUS "${_gpu_report}\nDevice qualification requires executing the built acceptance case.")
endif()
file(READ "${OpenPFC_BUILD_MANIFEST}" _manifest)
string(JSON _revision GET "${_manifest}" source_revision)
string(JSON _state GET "${_manifest}" source_state)
file(SHA256 "${OpenPFC_BUILD_MANIFEST}" _manifest_sha)
# Content identity also works in an exported source archive.
set(_source_hashes "")
foreach(_file CMakeLists.txt main.cpp case.hpp identity.hpp.in)
  file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/${_file}" _hash)
  string(APPEND _source_hashes "${_file}:${_hash}\n")
endforeach()
string(SHA256 _consumer_sha "${_source_hashes}")
string(JSON _heffte GET "${_manifest}" capabilities heffte)
if(NOT _heffte)
  message(FATAL_ERROR "NOT_QUALIFIED: distributed FFT requires a HeFFTe SDK")
endif()
set(_compiler_language "${_backend}")
if(backend STREQUAL "cpu")
  set(_compiler_language CXX)
endif()
set(_consumer_compiler "${CMAKE_${_compiler_language}_COMPILER_ID} ${CMAKE_${_compiler_language}_COMPILER_VERSION}")
set(_consumer_host_compiler "${CMAKE_CXX_COMPILER_ID} ${CMAKE_CXX_COMPILER_VERSION}")
configure_file(identity.hpp.in identity.hpp @ONLY)
add_executable(backend_acceptance main.cpp)
if(backend STREQUAL "cuda")
  set_source_files_properties(main.cpp PROPERTIES LANGUAGE CUDA)
  target_compile_definitions(backend_acceptance PRIVATE CHECK_CUDA)
  target_compile_options(backend_acceptance PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false;-std=c++20>)
elseif(backend STREQUAL "hip")
  # FindMPI intentionally omits include dirs when MPI_CXX_COMPILER is the
  # host compiler itself. Its explicitly queried compiler dirs are still
  # needed by HIP Clang, which is a different compiler.
  if(MPI_CXX_COMPILER_INCLUDE_DIRS)
    target_include_directories(backend_acceptance SYSTEM PRIVATE ${MPI_CXX_COMPILER_INCLUDE_DIRS})
  endif()
  set_source_files_properties(main.cpp PROPERTIES LANGUAGE HIP)
  target_compile_definitions(backend_acceptance PRIVATE CHECK_HIP)
endif()
target_include_directories(backend_acceptance PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(backend_acceptance PRIVATE OpenPFC::openpfc)
target_compile_features(backend_acceptance PRIVATE cxx_std_20)
set_target_properties(backend_acceptance PROPERTIES CUDA_STANDARD 20 HIP_STANDARD 20)
# Device capability is mandatory; never substitute a host FFT.
if(NOT backend STREQUAL "cpu")
  string(JSON _recorded_architectures GET "${_manifest}" toolchains "${_backend}" architectures)
  foreach(_target IN LISTS CMAKE_${_backend}_ARCHITECTURES)
    string(REGEX REPLACE "-(real|virtual)$" "" _target "${_target}")
    if(NOT _target IN_LIST _recorded_architectures)
      message(FATAL_ERROR "NOT_QUALIFIED: consumer target ${_target} is not an exact SDK target (${_recorded_architectures})")
    endif()
  endforeach()
  string(JSON _spectral GET "${_manifest}" capabilities "${backend}_spectral")
  string(JSON _device_buffers GET "${_manifest}" mpi "${_backend}_device_buffers")
  if(NOT _device_buffers)
    message(FATAL_ERROR "NOT_QUALIFIED: SDK device-buffer MPI transport is disabled")
  endif()
  if(NOT _spectral)
    message(FATAL_ERROR "NOT_QUALIFIED: selected SDK has no ${backend} spectral backend")
  endif()
endif()
